/* ========= HELPERS ========= */

function isDependentFollowup(
text: string
): boolean {
const t =
text
.trim()
.toLowerCase();

const contextualPatterns =
/po co|dlaczego|czemu|i co|czyli|serio|dokładnie|właśnie|rozwiń|wyjaśnij|tak naprawdę/i;

const vagueReferences =
/\b(to|tego|tym|taka|takie|tak)\b/i;

return (
t.length < 180 &&
(
contextualPatterns.test(t) ||
vagueReferences.test(t)
)
);
}

function buildFollowupHint(
userText: string,
lastAssistant: string
) {
const t =
userText
.trim()
.toLowerCase();

if (
/^(ale )?(po co|dlaczego|czemu)??$/.test(
t
)
) {
return `
User asks about the REASON
behind the previous explanation.

Continue the same topic.
Do not reset conversation.
Explain the mechanism directly.
`;
}

if (
/nie rozumiem|bez sensu|co?/.test(
t
)
) {
return `
User did not understand
the previous explanation.

Explain the same idea
more simply.
`;
}

return `
This is a continuation
of the previous topic.

Continue naturally.
Do not restart context.
`;
}

function extractCoreProblem(
  text: string,
  history: ChatMsg[]
) {
  const full =
    [
      ...history.map(
        (m) => m.content
      ),
      text,
    ]
      .join(" ")
      .toLowerCase();

  const topics = [
    {
      key: "children",
      label:
        "brak kontaktu z dziećmi",

      regex:
        /dzieci|kontakt z dziecmi/g,
    },

    {
      key: "relationship",
      label:
        "rozpad związku",

      regex:
        /zona|żona|rozstanie|odeszla|odeszła/g,
    },

    {
      key: "mental",
      label:
        "przeciążenie / depresja",

      regex:
        /depresja|nie dam rady|rozsypany|wykonczony|wykończony|nie moge zyc/g,
    },
  ];

  const scored =
    topics.map((t) => {
      const matches =
        full.match(
          t.regex
        );

      return {
        ...t,

        score:
          matches
            ? matches.length
            : 0,
      };
    });

  scored.sort(
    (
      a,
      b
    ) =>
      b.score -
      a.score
  );

  return {
    main:
      scored[0]?.score > 0
        ? scored[0]
        : null,

    secondary:
      scored[1]?.score > 0
        ? scored[1]
        : null,

    all:
      scored.filter(
        (s) =>
          s.score > 0
      ),
  };
}
